\begin{frame}{Why not VMs?}
\begin{itemize}
\item<1-> VM images might be big
\item<2-> VM builds might be complicated
\item<3-> significant overhead
\item<4-> getting \& running VM images might be annoying
\end{itemize}
\end{frame}

\begin{frame}{Solving these Problems with Docker}
\begin{itemize}
\item<1-> Docker Engine runs containerized, isolated applications
\item<2-> standarized \textbf{Dockerfile} syntax defines "VM" Images
\item<3-> a Docker client pulls, configures \& runs the images
\end{itemize}
\end{frame}

\begin{frame}{Kubernetes}
\begin{itemize}
	\item<1-> ..automatically scales stateless container loads
	\item<2-> ..automatically handles certificates
	\item<3-> ..has great UIs for semi-technical operators
	\item<4-> ..automatically encrypts \& handles internal traffic
	\item<5-> ..has great automatic deployment options (GitOps)
\end{itemize}
\end{frame}


\begin{frame}{Kubernetes also}
	\begin{itemize}
		\item<1-> ..hard to maintain by a single person
		\item<2-> ..very expensive if you don't self-host it
		\item<3-> ..does not make your application magically HA (Highly Available)
		\item<4-> ..does not allow easy manual intervention
%		\item<5-> ..requires you to think in certain frameworks
%		\item<6-> ..is worse the less stateless your Pod is
	\end{itemize}
